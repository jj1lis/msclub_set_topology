\begin{frame}[fragile]{\S 1 集合とは}
    \begin{block}{集合}
        「もの」の集まりで，\underline{その範囲がはっきり定まっている}もの
    \end{block}
    あるもの$a$と集まり$A$が与えられたとき，$a$が$A$に入っているかを主観なく判断できるか？
    \begin{itemize}
        \item 「自然数全体」や「日本人全体」は集合である
        \item 「大きい自然数全体」や「美人全体」は集合でない 
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{\S 1 集合とは}
    \begin{block}{記法}
        $A$を集合とする．
        \begin{itemize}
            \item $A$を構成する「もの」一つ一つを，$A$の\alert{元}または\alert{要素}という．
            \item $a$が$A$の元であるとき，$a$は$A$に\alert{属する}ないし\alert{含まれる}といい，$a \in A$で表す．
            \item $a$が$A$の元でないとき，$a \notin A$と書く（以後，他の記法も同様）．
        \end{itemize}
    \end{block}
    $a \in A$は向きを反転させて$A \ni a$とも書ける．

    \begin{block}{記法}
        集合$A$と$B$が同じ構成要素から成るとき，$A = B$と書く．
    \end{block}
    \medskip

    $A = B$であることは，言い換えれば
    \begin{itemize}
        \item $x$が$A$の元であるならば，$x$が$B$の元でもあること
        \item $x$が$B$の元であるならば，$x$が$A$の元でもあること
    \end{itemize}
    がそれぞれ成り立つことである．

\begin{frame}[fragile]{\S 1 集合とは}
    \begin{block}{集合の表し方}
        集合を表すための2通りの表記を導入する．
        \begin{enumerate}
            \item 元をすべて列挙して波括弧で囲う
                \begin{itemize}
                    \item[ex.] 自然数2, 5, 7 から成る集合は$\qty{2, 5, 7}$
                \end{itemize}
            \item ある条件$P$を満たすもの全体$x$として集合を定め，$\Set{x | P(x)}$で表す
                \begin{itemize}
                    \item[ex.] 1億以下の自然数全体は$\Set{x | \text{$x$は自然数，$x \le 10000000$}}$
                        \begin{itemize}
                            \item $\Set{x \in \Natural | x \le 100000000}$とも書く
                    \end{itemize}
                \end{itemize}
        \end{enumerate}
    \end{block}
    \medskip

    一般的には，これらの表記はそれぞれ
    \begin{enumerate}
        \item 外延的表記
        \item 内包的表記
    \end{enumerate}
    と呼ばれることが多い．
\end{frame}

\begin{frame}[fragile]{\S 1 集合とは}
    \begin{block}{部分集合}
        $A, B$を集合とする．
        \begin{itemize}
            \item $A$の任意の元が$B$に含まれるとき，$A$は$B$の\alert{部分集合}であるといい，\\
                $A \subset B$や$B \supset A$と書く．
            \item $A$は$B$に\alert{包まれる}とも言う．
            \item 特に$A \subset B$かつ$A \neq B$であるとき，$A$は$B$の\alert{真部分集合}であるといい，
                $A \subsetneqq B$と書く．
        \end{itemize}
    \end{block}

    \begin{itemize}
        \item $A \subset B$を明示的に$A \subseteqq B$と書く流儀もある
    \end{itemize}

    \begin{block}{性質}
        \begin{enumerate}
            \item $A \subset A$である．
            \item $A \subset B$かつ$B \subset A$ならば$A = B$である．
            \item $A \subset B$かつ$B \subset C$ならば$A \subset C$である．
        \end{enumerate}
    \end{block}

\end{frame}

\begin{frame}[fragile]{\S 1 集合とは}
    \begin{block}{部分集合}
        $A \subset B\ \define$\ $A$の任意の元が$B$に含まれる
    \end{block}
    \medskip
    
    空集合$\varnothing$は，任意の集合の部分集合である．
    \begin{itemize}
        \item 対偶律より「$x \in A$ならば$x \in B$」は「$x \notin B$ならば$x \notin A$」と同値
        \item どんな$B$と$x \notin B$を取っても$x \notin \varnothing$を満たすので，$\emptyset \subset B$が言える
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{\S 1 集合とは}
    \begin{block}{巾集合}
        集合$A$の部分集合全体を$A$の\alert{巾集合}といい，$\power(A)$で表す．
    \end{block}
    \medskip
    
    空集合$\varnothing$は，任意の集合の部分集合である．
    \begin{itemize}
        \item 対偶律より「$x \in A$ならば$x \in B$」は「$x \notin B$ならば$x \notin A$」と同値
        \item どんな$B$と$x \notin B$を取っても$x \notin \varnothing$を満たすので，$\emptyset \subset B$が言える
    \end{itemize}
    \bigskip

    一般に，その元がふたたび集合であるような集合を\alert{集合族}と呼ぶ．
\end{frame}

\begin{frame}[fragile]{\S 2 集合の演算}
    \begin{block}{集合の演算}
        $A, B$を集合とする．
        \begin{itemize}
            \item $A$と$B$の元を寄せ集めてできる集合を$A$と$B$の\alert{和集合}といい，$A \cup B$で表す．
            \item $A$にも$B$にも含まれる元から成る集合を$A$と$B$の\alert{共通部分}といい，$A \cap B$で表す．
            \item $A$に含まれ，$B$に含まれない元から成る集合を$A$と$B$の\alert{差部分}といい，$A - B$で表す．
        \end{itemize}
    \end{block}

    $A \cap B$が空集合であるとき，$A$と$B$は\alert{交わらない}といい，さもなくば\alert{交わる}という．\\
    $A$と$B$が交わらないとき，$A \cup B$を$A$と$B$の\alert{非交和}ないし\alert{直和}という．
\end{frame}

\begin{frame}[fragile]{\S 2 集合の演算}
    \begin{block}{定理2.1}
        集合$A, B$について以下が成り立つ．
        \begin{columns}
            \begin{column}{0.5\linewidth}
                \begin{enumerate}[(1)]
                    \item $A \subset A \cup B$
                    \item $B \subset A \cup B$
                    \item $A \cup B = B \cup A$
                \end{enumerate}
            \end{column}
            \begin{column}{0.5\linewidth}
                \begin{enumerate}
                    \item[(4)] $A \cap B \subset A$
                    \item[(5)] $A \cap B \subset B$
                    \item[(6)] $A \cap B = B \cap A$
                \end{enumerate}
            \end{column}
        \end{columns}
    \end{block}
    \medskip
    \fbox{証明} \\
    \medskip

    (1) のみ示す．元$x \in A$を任意に取ると，定義より$x$は$A \cup B$の元でもある．\\したがって$A \subset A \cup B$．\qed

    \medskip
    \begin{block}{部分集合}
        $A \subset B\ \define$\ $A$の任意の元が$B$に含まれる
    \end{block}
\end{frame}

\begin{frame}[fragile]{\S 2 集合の演算}
    \begin{block}{定理2.2}
        集合$A, B, C, D$について以下が成り立つ．
        \begin{enumerate}[(1)]
            \item $A \subset C$かつ$B \subset C$ならば，$A \cup B \subset C$である．
            \item $D \subset A$かつ$D \subset B$ならば，$D \subset A \cap B$である．
        \end{enumerate}
    \end{block}
    \medskip
    \fbox{証明} \\
    \medskip
    (1) のみ示す．$x \in A \cup B$を任意に取ると，$x$は$A$の元であるか，または$B$の元である．\\
    どちらの場合にも$x \in C$が成り立つから，$A \cup B \subset C$．\qed

    \begin{block}{定理2.3（結合法則）}
        集合$A, B, C$について以下が成り立つ．
        \begin{enumerate}[(1)]
            \item $(A \cup B) \cup C = A \cup (B \cup C)$
            \item $(A \cap B) \cap C = A \cap (B \cap C)$
        \end{enumerate}
    \end{block}
\end{frame}

            \item[(6)] $A \cap B = B \cap A$
        \end{enumerate}
    \end{block}

\begin{frame}[fragile]{\S 2 集合の演算}
    \begin{block}{定理2.4（分配法則）}
        集合$A, B, C$について以下が成り立つ．
        \begin{enumerate}[(1)]
            \item $A \cup (B \cap C) = (A \cup B) \cap (A \cup C)$
            \item $A \cap (B \cup C) = (A \cap B) \cup (A \cap C)$
        \end{enumerate}
    \end{block}
    \smallskip
    \fbox{証明} (1) のみ示す．\\
    \smallskip
    \begin{description}[($\subset$)]
        \item[($\subset$)] $x \in A \cup (B \cap C)$と仮定すると，以下の2通りがありうる．%そのどちらにおいても$x \in (A \cup B) \cap (A \cup C)$となることを示す．
            \begin{description}[$x \in B \cap C$:]
                \item[$x \in A$:] 直ちに$x \in A \cup B$と$x \in A \cup C$が従うので，$x \in (A \cup B) \cap (A \cup C)$．
                \item[$x \in B \cap C$:] $x \in B$より$x \in A \cup B$．$x \in C$より$x \in A \cup C$．よって$x \in (A \cup B) \cap (A \cup C)$．
            \end{description}
        \item[($\supset$)] $x \in (A \cup B) \cap (A \cup C)$と仮定する．このとき以下の4通りが考えられる:\\
            (a) $x \in A$, $x \in A$\ (b) $x \in A$, $x \in C$\ (c) $x \in B$, $x \in A$ (d) $x \in B$, $x \in C$. \\
            (a), (b), (c) の場合は$x \in A$，(d) の場合は $x \in B \cap C$であるから，$x \in A \cup (B \cap C)$．\qed
    \end{description}
\end{frame}

\begin{frame}[fragile]{\S 3 ド・モルガンの法則}
    \begin{block}{定理3.1（ド・モルガンの法則）}
        集合$X, A, B$について以下が成り立つ．
        \begin{enumerate}[(1)]
            \item $X - (A \cup B) = (X - A) \cap (X - B)$
            \item $X - (A \cap B) = (X - A) \cup (X - B)$
        \end{enumerate}
    \end{block}
    \smallskip
    \fbox{証明} (1) のみ示す．\\
    \smallskip
    \begin{description}[($\subset$)]
        \item[($\subset$)] $x \in X - (A \cup B)$と仮定する．すなわち$x \notin A \cup B$であるから，$x \notin A$かつ$x \notin B$，\\
            また$x \in X$であるから$x \in X - A$かつ$x \in X - B$．よって$x \in (X - A) \cap (X - B)$．
        \item[($\supset$)] 同様．\qed
    \end{description}
\end{frame}
